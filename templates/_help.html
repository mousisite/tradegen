{# Help beside a figure, in the words of plain.HELP, which the page passes as help_text with its own settings filled in. A details element, so it opens and closes with no script; its summary is a real button to a keyboard and a screen reader, named "About ...". Import it with context. A key with no words renders nothing rather than breaking the page; test_help.py checks that every key used has words. #} {% macro help(key) -%} {%- set h = help_text.get(key) if help_text else none -%} {%- if h -%}
?About {{ h.name }}{{ h.text }}
{%- endif -%} {%- endmacro %} {# Several figures behind one "?", for the columns of a table, with the label said beside it. #} {% macro help_list(keys, label) -%} {%- set found = keys|select('in', help_text or {})|list -%} {%- if found -%}
?{{ label }}
{%- for key in found %}{% set h = help_text[key] %}
{{ h.name[:1]|upper ~ h.name[1:] }}
{{ h.text }}
{% endfor -%}
{%- endif -%} {%- endmacro %}
